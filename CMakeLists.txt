cmake_minimum_required(VERSION 3.20)
project(SolarSystemSim VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MARS_BUILD_APP "Build the SDL3/OpenGL desktop application" ON)

add_library(mars_terrain src/terrain.cpp src/satellites.cpp src/faceted.cpp src/mercury.cpp src/moon.cpp src/preview.cpp)
target_include_directories(mars_terrain PUBLIC src)
add_executable(mola_convert tools/mola_convert.cpp)
target_link_libraries(mola_convert PRIVATE mars_terrain)
add_executable(mercury_convert tools/mercury_convert.cpp)
target_link_libraries(mercury_convert PRIVATE mars_terrain)
add_executable(moon_convert tools/moon_convert.cpp)
target_link_libraries(moon_convert PRIVATE mars_terrain)
add_executable(preview_generate tools/preview_generate.cpp)
target_link_libraries(preview_generate PRIVATE mars_terrain)

if(MARS_BUILD_APP)
  find_package(SDL3 3.2 REQUIRED CONFIG)
  add_executable(solar_system_sim src/main.cpp src/renderer_belts.cpp src/renderer_sun.cpp src/renderer_space.cpp src/saturn.cpp src/saturn_view.cpp src/renderer_saturn.cpp src/satellites_view.cpp src/renderer_satellites.cpp src/renderer.cpp src/renderer_navigation.cpp src/renderer_guide.cpp src/guide.cpp src/renderer_mercury.cpp src/mercury_view.cpp src/moon_view.cpp src/renderer_moon.cpp src/io.cpp src/io_view.cpp src/renderer_io.cpp src/callisto.cpp src/callisto_view.cpp src/renderer_callisto.cpp src/venus.cpp src/venus_view.cpp src/renderer_venus.cpp src/renderer_earth.cpp src/earth.cpp src/earth_view.cpp src/renderer_jupiter.cpp src/jupiter.cpp src/jupiter_view.cpp src/environment.cpp src/orbit.cpp src/mars_view.cpp src/catalog.cpp src/navigation.cpp src/placeholders.cpp)
  target_link_libraries(solar_system_sim PRIVATE mars_terrain SDL3::SDL3)
  include(cmake/Packaging.cmake)
  add_custom_command(TARGET solar_system_sim POST_BUILD COMMAND ${CMAKE_COMMAND} -E
    copy_directory ${CMAKE_CURRENT_SOURCE_DIR}/assets $<TARGET_FILE_DIR:solar_system_sim>/assets)
  if(WIN32 AND TARGET SDL3::SDL3-shared)
    add_custom_command(TARGET solar_system_sim POST_BUILD COMMAND ${CMAKE_COMMAND} -E
      copy_if_different $<TARGET_FILE:SDL3::SDL3-shared> $<TARGET_FILE_DIR:solar_system_sim>)
  endif()
endif()
